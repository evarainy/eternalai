## 3. Mock Adapter Changes

Add `set_state` / `reset_state` to each Mock Adapter. The existing `execute()` logic is NOT removed — state-based data takes precedence when state is set, otherwise falls back to current behavior.

### 3.1 Interface pattern (apply to all 3 adapters)

```python
from __future__ import annotations
from typing import Any

class MockXxxAdapter:
    def __init__(self) -> None:
        self._mock_state: dict[str, Any] = {}

    def set_state(self, state: dict[str, Any]) -> None:
        self._mock_state = state

    def reset_state(self) -> None:
        self._mock_state = {}

    async def execute(self, capability_id: str, arguments: dict[str, Any], execution_context: dict[str, Any]) -> AdapterResult:
        # existing error_mode check FIRST (unchanged)
        error_mode = execution_context.get("mock_error_mode")
        if error_mode is not None:
            return AdapterResult(status="error", data=None, error_code=MOCK_ERROR_MODE_TO_ERROR_CODE.get(error_mode))

        # NEW: if state is set, use it
        if self._mock_state:
            return self._build_from_state(capability_id, arguments)

        # existing hardcoded fallback (unchanged)
        ...
```

### 3.2 MockOAAdapter state resolution (_build_from_state)

State keys (from GT fixtures):
- `pending_workflows`: list[dict] → return as `data={"workflows": state["pending_workflows"]}`
- `workflow_status`: dict → return as `data=state["workflow_status"]`
- `submit_result`: dict → return as `data=state["submit_result"]`

Resolution logic: check capability_id contains "list_pending" → use `pending_workflows`; contains "get_workflow" or "workflow_status" → use `workflow_status`; contains "submit" → use `submit_result`; default: return `data=self._mock_state`.

Always return `AdapterResult(status="success", data=...)`.

If state key for the capability branch is missing, return `AdapterResult(status="error", data=None, error_code="adapter_state_missing")`.

### 3.3 MockU8Adapter state resolution (_build_from_state)

State keys:
- `document`: dict → use when capability_id contains "document"
- `vendor_balance`: dict → use when capability_id contains "vendor" or "balance"

If key missing for branch: `AdapterResult(status="error", data=None, error_code="adapter_state_missing")`.

### 3.4 MockHikvisionIVMSAdapter state resolution (_build_from_state)

State keys:
- `device`: dict → use for any capability (only one shape in fixtures)

If `device` key missing: `AdapterResult(status="error", data=None, error_code="adapter_state_missing")`.

"""Verify mock state injection and isolation."""
import asyncio
from tests.golden_tasks.conftest import load_fixture

def test_gt001_oa_state_can_be_injected_and_returned(oa_adapter) -> None:
    fixture = load_fixture("GT-001")
    mock_state = fixture["given"]["mock_oa_state"]
    oa_adapter.set_state(mock_state)
    result = asyncio.run(oa_adapter.execute("oa.list_pending_workflows", {}, {}))
    assert result.status == "success"
    assert result.data is not None
    workflows = result.data.get("workflows") or result.data.get("pending_workflows")
    assert isinstance(workflows, list)
    assert len(workflows) == 3

def test_gt001_oa_state_reset_clears_injected_data(oa_adapter) -> None:
    fixture = load_fixture("GT-001")
    oa_adapter.set_state(fixture["given"]["mock_oa_state"])
    oa_adapter.reset_state()
    # After reset, falls back to hardcoded behavior (still returns success)
    result = asyncio.run(oa_adapter.execute("oa.list_pending_workflows", {}, {}))
    assert result.status == "success"
    # Hardcoded fallback does not have the 3-item pending_workflows list
    data_str = str(result.data)
    assert "OA-WF-2026-0001" not in data_str

def test_two_successive_gt_loads_do_not_leak_state(oa_adapter) -> None:
    # Load GT-001 state, then reset, then load GT-002 state
    gt001 = load_fixture("GT-001")
    gt002 = load_fixture("GT-002")

    oa_adapter.set_state(gt001["given"]["mock_oa_state"])
    oa_adapter.reset_state()
    oa_adapter.set_state(gt002["given"]["mock_oa_state"])

    assert result.status == "success"
    assert result.data is not None
    # Should return GT-002's workflow_status data, not GT-001's pending_workflows

    assert "approved" in data_str
